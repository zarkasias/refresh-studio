(() => {
  const c = document.createElement('canvas').getContext('2d', { willReadFrequently: true });
  const hex = v => {
    c.clearRect(0, 0, 1, 1); c.fillStyle = '#000'; c.fillStyle = v; c.fillRect(0, 0, 1, 1);
    const [r, g, b, a] = c.getImageData(0, 0, 1, 1).data;
    if (a < 128) return null;
    return '#' + [r, g, b].map(n => n.toString(16).padStart(2, '0')).join('');
  };
  const bg = {}, text = {}, fonts = {};
  for (const el of document.querySelectorAll('html, body, body *')) {
    const r = el.getBoundingClientRect();
    if (!r.width || !r.height) continue;
    const s = getComputedStyle(el);
    const b = hex(s.backgroundColor);
    if (b) bg[b] = (bg[b] || 0) + Math.round(r.width * r.height);
    const hasText = [...el.childNodes].some(n => n.nodeType === 3 && n.textContent.trim());
    if (hasText) {
      const t = hex(s.color);
      if (t) text[t] = (text[t] || 0) + 1;
      const f = s.fontFamily.split(',')[0].replace(/["']/g, '').trim();
      fonts[f] = (fonts[f] || 0) + 1;
    }
  }
  const top = (o, n) => Object.entries(o).sort((a, b) => b[1] - a[1]).slice(0, n);
  console.log('Backgrounds (by area)'); console.table(top(bg, 6));
  console.log('Text colors (by count)'); console.table(top(text, 6));
  console.log('Fonts'); console.table(top(fonts, 4));
})();
